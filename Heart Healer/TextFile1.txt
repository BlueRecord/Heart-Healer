// 2. DrawCard (카드를 한 장 뽑음)
void DrawCard()
{
    // [예외 처리] 3. 만약 드로우 덱이 비어있다면 무덤에서 리필
    if (drawDeck.Count == 0)
    {
        RefillDeckFromGrave();
    }

    // 만약 무덤도 비어서 진짜 뽑을 카드가 아예 없다면 리턴
    if (drawDeck.Count == 0) return;

    // 맨 위의 카드 한 장 지정 (0번 인덱스)
    CardData drawnCard = drawDeck[0];
    drawDeck.RemoveAt(0); // 덱에서 제거

    // 핸드가 가득 찼는지 체크
    if (hand.Count < maxHandSize)
    {
        hand.Add(drawnCard); // 핸드 데이터에 추가
        CardUI.CreateCardMotion(drawnCard); // UI에 드로우 연출 실행 명령
    }
    else
    {
        grave.Add(drawnCard); // 핸드가 꽉 찼으면 바로 무덤으로
        Debug.Log("손패가 가득 차 카드가 무덤으로 버려졌습니다.");
    }
}

// 3. RefillDeckFromGrave (덱이 비었을 때 무덤을 회수)
void RefillDeckFromGrave()
{
    // 무덤에 있는 모든 데이터를 드로우 덱으로 이동
    drawDeck.AddRange(grave);
    grave.Clear(); // 무덤 비우기
    
    // 다시 리필되었으니 반드시 섞어줌
    Shuffle();
    Debug.Log("무덤의 카드를 덱으로 되돌리고 셔플했습니다.");
}
